cmake_minimum_required(VERSION 3.20)
project(visual_odometry VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rgbd_camera_interfaces 0.3.0 REQUIRED)
find_package(visual_odometry_interfaces REQUIRED)
find_package(isaac_ros_visual_slam_interfaces REQUIRED)
add_library(odometry_core src/core.cpp)
target_include_directories(odometry_core PUBLIC include)
ament_target_dependencies(odometry_core PUBLIC nav_msgs rgbd_camera_interfaces visual_odometry_interfaces)
add_executable(odometry_adapter src/adapter.cpp)
target_link_libraries(odometry_adapter PRIVATE odometry_core)
ament_target_dependencies(odometry_adapter PUBLIC rclcpp rcl_interfaces isaac_ros_visual_slam_interfaces)
install(TARGETS odometry_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/odometry_supervisor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test_core test/test_core.cpp)
  target_link_libraries(test_core PRIVATE odometry_core)
  add_test(NAME odometry_contract COMMAND test_core)
  add_test(NAME odometry_process_lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_supervisor.py)
  set_tests_properties(odometry_process_lifecycle PROPERTIES TIMEOUT 15)
endif()
ament_package()
